   118→// ─── Helpers ─────────────────────────────────────────────────────────────
   119→
   120→function formatDuration(seconds: number): string {
   121→  const m = Math.floor(seconds / 60);
   122→  const s = seconds % 60;
   123→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   124→}
   125→
   126→function formatDate(dateStr: string): string {
   127→  try {
   128→    return new Date(dateStr).toLocaleDateString('fr-FR', {
   129→      day: 'numeric',
   130→      month: 'short',
   131→      year: 'numeric',
   132→    });
   133→  } catch {
   134→    return dateStr;
   135→  }
   136→}
   137→
   138→function getBarColor(passRate: number): string {
   139→  if (passRate >= 80) return 'bg-emerald-500';
   140→  if (passRate >= 60) return 'bg-amber-500';
   141→  return 'bg-red-500';
   142→}
   143→
   144→function getBarTextColor(passRate: number): string {
   145→  if (passRate >= 80) return 'text-emerald-400';
   146→  if (passRate >= 60) return 'text-amber-400';
   147→  return 'text-red-400';
   148→}
