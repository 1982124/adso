   151→// ─── Circular Progress ───────────────────────────────────────
   152→function CircularProgress({ value, max, size = 120, strokeWidth = 10, color = '#10b981' }: {
   153→  value: number; max: number; size?: number; strokeWidth?: number; color?: string
   154→}) {
   155→  const radius = (size - strokeWidth) / 2
   156→  const circumference = 2 * Math.PI * radius
   157→   const offset = circumference - (value / max) * circumference
   158→
   159→  return (
   160→    <svg width={size} height={size} className="-rotate-90">
   161→      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#1e293b" strokeWidth={strokeWidth} />
   162→      <motion.circle
   163→        cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color}
   164→        strokeWidth={strokeWidth} strokeLinecap="round"
   165→        strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }}
   166→        animate={{ strokeDashoffset: offset }} transition={{ duration: 1.2, ease: 'easeOut' }}
   167→      />
   168→    </svg>
   169→  )
   170→}

   674→// ─── Score Card Sub-component ────────────────────────────────
   675→function ScoreCard({ label, value, threshold, critical, unit = '', icon }: {
   676→  label: string; value: number; threshold: number; critical?: boolean; unit?: string; icon: React.ReactNode
   677→}) {
   678→  const exceeds = critical ? value > threshold : value > threshold
   679→  const color = critical
   680→    ? (exceeds ? 'text-red-400' : 'text-emerald-400')
   681→    : (exceeds ? 'text-yellow-400' : 'text-emerald-400')
   682→  const bgColor = critical
   683→    ? (exceeds ? 'bg-red-500/10' : 'bg-emerald-500/10')
   684→    : (exceeds ? 'bg-yellow-500/10' : 'bg-emerald-500/10')
   685→
   686→  return (
   687→    <Card className="bg-slate-900 border-slate-800">
   688→      <CardContent className="pt-5 pb-4 space-y-2">
   689→        <div className="flex items-center gap-2">
   690→          <span className={color}>{icon}</span>
   691→          <span className="text-slate-400 text-xs uppercase tracking-wider">{label}</span>
   692→        </div>
   693→        <p className={`text-2xl font-bold ${color}`}>
   694→          {value}
   695→          {unit && <span className="text-sm text-slate-500 ml-1">{unit}</span>}
   696→        </p>
   697→        <p className={`text-xs ${exceeds ? (critical ? 'text-red-400/70' : 'text-yellow-400/70') : 'text-slate-600'}`}>
   698→          {exceeds ? `Seuil: > ${threshold}${unit}` : 'Dans la norme'}
   699→        </p>
   700→      </CardContent>
   701→    </Card>
   702→  )
   703→}
