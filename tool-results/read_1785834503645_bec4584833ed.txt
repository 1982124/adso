   480→   480→function shuffleArray<T>(arr: T[]): T[] {
   481→   481→  const shuffled = [...arr];
   482→   482→  for (let i = shuffled.length - 1; i > 0; i--) {
   483→   483→    const j = Math.floor(Math.random() * (i + 1));
   484→   484→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
   485→   485→  }
   486→   486→  return shuffled;
   487→   487→}
