   482→function shuffleArray<T>(arr: T[]): T[] {
   483→  const shuffled = [...arr];
   484→  for (let i = shuffled.length - 1; i > 0; i--) {
   485→    const j = Math.floor(Math.random() * (i + 1));
   486→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
   487→  }
   488→  return shuffled;
   489→}

   559→        <div className={`${baseClasses} h-16 w-16 ${signColor} rounded-full`}>{children}</div>
