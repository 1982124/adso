   402→// ─── Animated Number ─────────────────────────────────────────
   403→function AnimatedNumber({ value, className = '' }: { value: number; className?: string }) {
   404→  const [display, setDisplay] = useState(0)
   405→  useEffect(() => {
   406→    let start = 0
   407→    const end = value
   408→    const duration = 1200
   409→    const step = (end - start) / (duration / 16)
   410→    const timer = setInterval(() => {
   411→      start += step
   412→      if ((step > 0 && start >= end) || (step < 0 && start <= end)) {
   413→        setDisplay(end)
   414→        clearInterval(timer)
   415→      } else {
   416→        setDisplay(Math.round(start))
   417→      }
   418→    }, 16)
   419→    return () => clearInterval(timer)
   420→  }, [value])
   421→  return <span className={className}>{display}</span>
   422→}
