   180→// ═══════════════════════════════════════════════════════════
   181→// Helpers
   182→// ═══════════════════════════════════════════════════════════
   183→
   184→function clamp(val: number, min: number, max: number) {
   185→  return Math.min(max, Math.max(min, val));
   186→}
   187→
   188→function getStatus(value: number, normalRange: [number, number]): 'normal' | 'warning' | 'critical' {
   189→  if (value >= normalRange[0] && value <= normalRange[1]) return 'normal';
   190→  const distLow = normalRange[0] - value;
   191→  const distHigh = value - normalRange[1];
   192→  const dist = Math.max(distLow, distHigh);
   193→  const range = normalRange[1] - normalRange[0];
   194→  return dist > range * 0.5 ? 'critical' : 'warning';
   195→}
   196→
   197→function getStatusColor(status: 'normal' | 'warning' | 'critical') {
   198→  switch (status) {
   199→    case 'normal': return 'bg-emerald-500';
   200→    case 'warning': return 'bg-amber-500';
   201→    case 'critical': return 'bg-red-500';
   202→  }
   203→}
   204→
   205→function getStatusTextColor(status: 'normal' | 'warning' | 'critical') {
   206→  switch (status) {
   207→    case 'normal': return 'text-emerald-400';
   208→    case 'warning': return 'text-amber-400';
   209→    case 'critical': return 'text-red-400';
   210→  }
   211→}
   212→
   213→function getDTCSeverityBadge(severity: string) {
   214→  switch (severity) {
   215→    case 'high':
   216→      return <Badge className='bg-red-500/20 text-red-400 border-red-500/30 hover:bg-red-500/30'>Critique</Badge>;
   217→    case 'medium':
   218→      return <Badge className='bg-amber-500/20 text-amber-400 border-amber-500/30 hover:bg-amber-500/30'>Modéré</Badge>;
   219→    case 'low':
   220→      return <Badge className='bg-emerald-500/20 text-emerald-400 border-emerald-500/30 hover:bg-emerald-500/30'>Mineur</Badge>;
   221→    default:
   222→      return <Badge variant='outline'>{severity}</Badge>;
   223→  }
   224→}
