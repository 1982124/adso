   166→            title: l.title as string,
   167→            category: l.category as Category,
   168→            categoryKey: l.categoryKey as string | undefined,
   169→            rating: l.rating as number,
   170→            reviews: l.reviews as number,
   171→            price: l.price as string,

   204→        s.location.toLowerCase().includes(searchQuery.toLowerCase())

   421→                  <motion.div
