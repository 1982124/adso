   151→   151→// ─── Circular Progress ───────────────────────────────────────
   152→   152→function CircularProgress({ value, max, size = 120, strokeWidth = 10, color = '#10b981' }: {
   153→   153→  value: number; max: number; size?: number; strokeWidth?: number; color?: string
   154→   154→}) {
   155→   155→  const radius = (size - strokeWidth) / 2
   156→   156→  const circumference = 2 * Math.PI * radius
   157→   157→   const offset = circumference - (value / max) * circumference
   158→   158→
   159→   159→  return (
   160→   160→    <svg width={size} height={size} className="-rotate-90">
   161→   161→      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#1e293b" strokeWidth={strokeWidth} />
   162→   162→      <motion.circle
   163→   163→        cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color}
   164→   164→        strokeWidth={strokeWidth} strokeLinecap="round"
   165→   165→        strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }}
   166→   166→        animate={{ strokeDashoffset: offset }} transition={{ duration: 1.2, ease: 'easeOut' }}
   167→   167→      />
   168→   168→    </svg>
   169→   169→  )
   170→   170→}
