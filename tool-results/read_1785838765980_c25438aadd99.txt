   403→// ─── Animated Number ─────────────────────────────────────────
   404→function AnimatedNumber({ value, className = '' }: { value: number; className?: string }) {
   405→  const [display, setDisplay] = useState(0)
   406→  useEffect(() => {
   407→    let start = 0
   408→    const end = value
   409→    const duration = 1200
   410→    const step = (end - start) / (duration / 16)
   411→    const timer = setInterval(() => {
   412→      start += step
   413→      if ((step > 0 && start >= end) || (step < 0 && start <= end)) {
   414→        setDisplay(end)
   415→        clearInterval(timer)
   416→      } else {
   417→        setDisplay(Math.round(start))
   418→      }
   419→    }, 16)
   420→    return () => clearInterval(timer)
   421→  }, [value])
   422→  return <span className={className}>{display}</span>
   423→}
