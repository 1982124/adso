   158→function formatTime(seconds: number): string {
   159→  const m = Math.floor(seconds / 60);
   160→  const s = seconds % 60;
   161→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;

   353→// ─── RUNNING STATE ─────────────────────────────────────────────────────────
   354→
   355→function RunningState({
   356→  questions,
   357→  examConfig,
   358→  timeRemaining,
   359→  selectedOption,
   360→  answered,
   361→  onSelectOption,
   362→  onNext,
   363→  onFinish,
   364→  currentIndex,
   365→  answers,
   366→}: {
   367→  questions: Question[];
   368→  examConfig: ExamConfig;
   369→  timeRemaining: number;
   370→  selectedOption: number | null;
   371→  answered: boolean;
   372→  onSelectOption: (index: number) => void;
   373→  onNext: () => void;
   374→  onFinish: () => void;
   375→  currentIndex: number;
   376→  answers: Map<string, number>;
   377→}) {
   378→  const q = questions[currentIndex];
   379→  const total = questions.length;
   380→  const progressPercent = Math.round(((currentIndex + (answered ? 1 : 0)) / total) * 100);
   381→  const isLowTime = timeRemaining > 0 && timeRemaining < 60;
   382→
   383→  return (
   384→    <div className="space-y-6">
   385→      {/* Top bar */}
   386→      <motion.div
   387→        initial={{ opacity: 0, y: -10 }}
   388→        animate={{ opacity: 1, y: 0 }}
   389→        className="flex flex-wrap items-center justify-between gap-3"
   390→      >
   391→        <div className="flex items-center gap-3">
   392→          <span className="text-slate-300 text-sm font-medium">
   393→            Question {currentIndex + 1}/{total}
   394→          </span>
   395→          <Badge
   396→            variant="outline"
   397→            className={`${examConfig.badgeColor} border-0 text-xs`}
   398→          >
   399→            {examConfig.badge}
   400→          </Badge>
   401→        </div>
   402→        {examConfig.duration > 0 && (
   403→          <div
   404→            className={`flex items-center gap-2 px-3 py-1.5 rounded-lg ${
   405→              isLowTime
   406→                ? 'bg-red-500/10 border border-red-500/30'
   407→                : 'bg-slate-800/60'
   408→            }`}
   409→          >
   410→            <Clock
   411→              className={`w-4 h-4 ${
   412→                isLowTime ? 'text-red-400' : 'text-slate-400'
   413→              }`}
   414→            />
   415→            <span
   416→              className={`font-mono text-sm font-semibold ${
   417→                isLowTime ? 'text-red-400' : 'text-white'
   418→              }`}
   419→            >
   420→              {formatTime(timeRemaining)}
   421→            </span>
   422→          </div>
   423→        )}
   424→      </motion.div>
   425→
   426→      {/* Progress bar */}
   427→      <Progress value={progressPercent} className="h-2 [&>div]:bg-emerald-500" />
   428→
   429→      {/* Question card */}
   430→      <AnimatePresence mode="wait">
   431→        <motion.div
   432→          key={q.id}
   433→          initial={{ opacity: 0, x: 30 }}
   434→          animate={{ opacity: 1, x: 0 }}
   435→          exit={{ opacity: 0, x: -30 }}
   436→          transition={{ duration: 0.3 }}
   437→        >
   438→          <Card className="bg-slate-900/80 border-slate-800/60">
   439→            <CardContent className="p-6 space-y-5">
   440→              {/* Badges */}
   441→              <div className="flex flex-wrap items-center gap-2">
   442→                <Badge variant="outline" className="bg-slate-800 text-slate-300 border-slate-700 text-xs">
   443→                  {CATEGORY_MAP[q.category] || q.category}
   444→                </Badge>
   445→                <Badge
   446→                  variant="outline"
   447→                  className={`${DIFFICULTY_BADGE[q.difficulty] || 'bg-slate-700 text-slate-300'} border-0 text-xs`}
   448→                >
   449→                  {DIFFICULTY_MAP[q.difficulty] || q.difficulty}
   450→                </Badge>
   451→                {q.theme && (
   452→                  <Badge variant="outline" className="bg-slate-800/60 text-slate-400 border-slate-700/50 text-xs">
   453→                    {q.theme}
   454→                  </Badge>
   455→                )}
   456→              </div>
   457→
   458→              {/* Question text */}
   459→              <p className="text-white text-base leading-relaxed font-medium">
   460→                {q.question}
   461→              </p>
   462→
   463→              {/* Options */}
   464→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   465→                {q.options.map((option, idx) => {
   466→                  const isCorrect = idx === q.correctIndex;
   467→                  const isSelected = selectedOption === idx;
   468→                  const showResult = answered;
   469→
   470→                  let borderColor = 'border-slate-800/60 hover:border-emerald-500/30';
   471→                  let bgColor = 'bg-slate-800/60 hover:bg-slate-700/40';
   472→
   473→                  if (showResult) {
   474→                    if (isCorrect) {
   475→                      borderColor = 'border-emerald-500';
   476→                      bgColor = 'bg-emerald-500/10';
   477→                    } else if (isSelected && !isCorrect) {
   478→                      borderColor = 'border-red-500';
   479→                      bgColor = 'bg-red-500/10';
   480→                    }
   481→                  } else if (isSelected) {
   482→                    borderColor = 'border-emerald-500/50';
   483→                    bgColor = 'bg-emerald-500/5';
   484→                  }
   485→
   486→                  return (
   487→                    <motion.div
   488→                      key={idx}
   489→                      whileHover={!answered ? { scale: 1.01 } : undefined}
   490→                      whileTap={!answered ? { scale: 0.99 } : undefined}
   491→                    >
   492→                      <div
   493→                        role="button"
   494→                        tabIndex={0}
   495→                        onKeyDown={(e) => {
   496→                          if (!answered && (e.key === 'Enter' || e.key === ' ')) {
   497→                            e.preventDefault();
   498→                            onSelectOption(idx);
   499→                          }
   500→                        }}
   501→                        onClick={() => !answered && onSelectOption(idx)}
   502→                        className={`flex items-start gap-3 p-4 rounded-xl border-2 cursor-pointer transition-all duration-200 ${borderColor} ${bgColor} ${
   503→                          !answered ? 'hover:shadow-[0_0_12px_rgba(16,185,129,0.15)]' : ''
   504→                        }`}
   505→                      >
   506→                        <span
   507→                          className={`w-8 h-8 rounded-lg flex items-center justify-center text-sm font-bold shrink-0 ${
   508→                            showResult && isCorrect
   509→                              ? 'bg-emerald-500 text-white'
   510→                              : showResult && isSelected && !isCorrect
   511→                                ? 'bg-red-500 text-white'
   512→                                : 'bg-slate-700 text-slate-300'
   513→                          }`}
   514→                        >
   515→                          {OPTION_LABELS[idx]}
   516→                        </span>
   517→                        <span
   518→                          className={`text-sm leading-relaxed pt-1 ${
   519→                            showResult && isCorrect
   520→                              ? 'text-emerald-300 font-medium'
   521→                              : showResult && isSelected && !isCorrect
   522→                                ? 'text-red-300'
   523→                                : 'text-slate-200'
   524→                          }`}
   525→                        >
   526→                          {option}
   527→                        </span>
   528→                        {showResult && isCorrect && (
   529→                          <CheckCircle2 className="w-5 h-5 text-emerald-400 shrink-0 ml-auto mt-0.5" />
   530→                        )}
   531→                        {showResult && isSelected && !isCorrect && (
   532→                          <XCircle className="w-5 h-5 text-red-400 shrink-0 ml-auto mt-0.5" />
   533→                        )}
   534→                      </div>
   535→                    </motion.div>
   536→                  );
   537→                })}
   538→              </div>
   539→
   540→              {/* Explanation */}
   541→              <AnimatePresence>
   542→                {answered && q.explanation && (
   543→                  <motion.div
   544→                    initial={{ opacity: 0, y: 10 }}
   545→                    animate={{ opacity: 1, y: 0 }}
   546→                    transition={{ duration: 0.3 }}
   547→                  >
   548→                    <div className="rounded-xl bg-slate-800/80 border border-slate-700/50 p-4 space-y-2">
   549→                      <div className="flex items-center gap-2">
   550→                        <AlertTriangle className="w-4 h-4 text-amber-400" />
   551→                        <span className="text-amber-300 text-sm font-semibold">
   552→                          Explication
   553→                        </span>
   554→                      </div>
   555→                      <p className="text-slate-300 text-sm leading-relaxed">
   556→                        {q.explanation}
   557→                      </p>
   558→                      {q.reference && (
   559→                        <p className="text-slate-500 text-xs mt-1">
   560→                          Référence : {q.reference}
   561→                        </p>
   562→                      )}
   563→                    </div>
   564→                  </motion.div>
   565→                )}
   566→              </AnimatePresence>
   567→
   568→              {/* Actions */}
   569→              <div className="flex items-center justify-between pt-2">
   570→                {answered && currentIndex < total - 1 && (
   571→                  <Button
   572→                    variant="outline"
   573→                    onClick={onNext}
   574→                    className="border-emerald-500/30 text-emerald-400 hover:bg-emerald-500/10 hover:text-emerald-300"
   575→                  >
   576→                    Question suivante
   577→                    <ChevronRight className="w-4 h-4 ml-1" />
   578→                  </Button>
   579→                )}
   580→                {(!answered || currentIndex === total - 1) && <div />}
   581→                <Button
   582→                  variant="outline"
   583→                  onClick={onFinish}
   584→                  className="border-red-500/30 text-red-400 hover:bg-red-500/10 hover:text-red-300"
   585→                >
   586→                  Terminer l&rsquo;examen
   587→                </Button>

   886→export default function ExamPlatform() {
   887→  const [examState, setExamState] = useState<ExamState>('idle');
   888→  const [examConfig, setExamConfig] = useState<ExamConfig>(EXAM_TYPES[0]);
   889→  const [questions, setQuestions] = useState<Question[]>([]);
   890→  const [currentIndex, setCurrentIndex] = useState(0);
   891→  const [answers, setAnswers] = useState<Map<string, number>>(new Map());
   892→  const [selectedOption, setSelectedOption] = useState<number | null>(null);
   893→  const [answered, setAnswered] = useState(false);
   894→  const [timeRemaining, setTimeRemaining] = useState(0);
   895→  const [result, setResult] = useState<ExamResult | null>(null);
   896→  const [elapsedSeconds, setElapsedSeconds] = useState(0);
   897→
   898→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   899→  const startTimeRef = useRef<number>(0);
   900→  const autoAdvanceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   901→
   902→  // Clear any pending auto-advance on unmount
   903→  useEffect(() => {
   904→    return () => {
   905→      if (timerRef.current) clearInterval(timerRef.current);
   906→      if (autoAdvanceRef.current) clearTimeout(autoAdvanceRef.current);
   907→    };
   908→  }, []);
   909→
   910→  // Timer countdown
   911→  useEffect(() => {
   912→    if (examState !== 'running' || examConfig.duration <= 0) return;
   913→
   914→    timerRef.current = setInterval(() => {
   915→      setTimeRemaining((prev) => {
   916→        if (prev <= 1) {
   917→          if (timerRef.current) clearInterval(timerRef.current);
   918→          return 0;
   919→        }
   920→        return prev - 1;
   921→      });
   922→    }, 1000);
   923→
   924→    return () => {
   925→      if (timerRef.current) clearInterval(timerRef.current);
   926→    };
   927→  }, [examState, examConfig.duration]);
   928→
   929→  // Auto-finish when timer reaches 0
   930→  useEffect(() => {
   931→    if (
   932→      examState === 'running' &&
   933→      examConfig.duration > 0 &&
   934→      timeRemaining === 0
   935→    ) {
   936→      handleFinish();
   937→    }
   938→  }, [timeRemaining, examState, examConfig.duration, handleFinish]);
   939→
   940→  // Elapsed time counter
   941→  useEffect(() => {
   942→    if (examState !== 'running') return;
   943→    startTimeRef.current = Date.now();
   944→
   945→    const interval = setInterval(() => {
   946→      setElapsedSeconds(
   947→        Math.floor((Date.now() - startTimeRef.current) / 1000)
   948→      );
   949→    }, 1000);
   950→
   951→    return () => clearInterval(interval);
   952→  }, [examState]);
   953→
   954→  // Post results to API
   955→  const submitResults = useCallback(
   956→    async (qs: Question[], ans: Map<string, number>, config: ExamConfig, elapsed: number) => {
   957→      const answerArray = Array.from(ans.entries()).map(([questionId, selectedOption]) => ({
   958→        questionId,
   959→        selectedOption,
   960→      }));
   961→
   962→      if (answerArray.length === 0) return;
   963→
   964→      try {
   965→        const res = await fetch('/api/exam', {
   966→          method: 'POST',
   967→          headers: { 'Content-Type': 'application/json' },
   968→          body: JSON.stringify({
   969→            userId: USER_ID,
   970→            answers: answerArray,
   971→            type: config.type,
   972→            duration: elapsed,
   973→            countryCode: 'FR',
   974→            licenseCode: 'B',
   975→          }),
   976→        });
   977→        if (res.ok) {
   978→          const data: ExamResult = await res.json();
   979→          setResult(data);
   980→        }
   981→      } catch (err) {
   982→        console.error('Erreur de soumission :', err);
   983→      }
   984→    },
   985→    []
   986→  );
   987→
   988→  const startExam = useCallback(
   989→    async (type: ExamType, difficulty: string, count: number) => {
   990→      const config = EXAM_TYPES.find((e) => e.type === type)!;
   991→      setExamConfig(config);
   992→      setExamState('loading');
   993→      setAnswers(new Map());
   994→      setCurrentIndex(0);
   995→      setSelectedOption(null);
   996→      setAnswered(false);
   997→      setResult(null);
   998→      setElapsedSeconds(0);
   999→
  1000→      // Build query params
  1001→      const params = new URLSearchParams();
  1002→      if (difficulty !== 'all') params.set('difficulty', difficulty);
  1003→      params.set('count', String(type === 'practice' ? count : config.questionCount));
  1004→
  1005→      try {
  1006→        const res = await fetch(`/api/learning/questions?${params.toString()}`);
  1007→        if (res.ok) {
  1008→          const data = await res.json();
  1009→          const fetchedQuestions: Question[] = data.questions;
  1010→          if (fetchedQuestions.length === 0) {
  1011→            setExamState('idle');
  1012→            return;
  1013→          }
  1014→          setQuestions(fetchedQuestions);
  1015→          setTimeRemaining(config.duration);
  1016→          setExamState('running');
  1017→        } else {
  1018→          setExamState('idle');
  1019→        }
  1020→      } catch (err) {
  1021→        console.error('Erreur de chargement des questions :', err);
  1022→        setExamState('idle');
  1023→      }
  1024→    },
  1025→    []
  1026→  );
  1027→
  1028→  const handleSelectOption = useCallback(
  1029→    (index: number) => {
  1030→      if (answered) return;
  1031→      const q = questions[currentIndex];
  1032→      setSelectedOption(index);
  1033→      setAnswered(true);
  1034→
  1035→      const newAnswers = new Map(answers);
  1036→      newAnswers.set(q.id, index);
  1037→      setAnswers(newAnswers);
  1038→
  1039→      // Auto-advance after 1.5s
  1040→      if (autoAdvanceRef.current) clearTimeout(autoAdvanceRef.current);
  1041→      autoAdvanceRef.current = setTimeout(() => {
  1042→        if (currentIndex < questions.length - 1) {
  1043→          setCurrentIndex((prev) => prev + 1);
  1044→          setSelectedOption(null);
  1045→          setAnswered(false);
  1046→        }
  1047→      }, 1500);
  1048→    },
  1049→    [answered, answers, currentIndex, questions]
  1050→  );
  1051→
  1052→  const handleNext = useCallback(() => {
  1053→    if (autoAdvanceRef.current) clearTimeout(autoAdvanceRef.current);
  1054→    if (currentIndex < questions.length - 1) {
  1055→      setCurrentIndex((prev) => prev + 1);
  1056→      setSelectedOption(null);
  1057→      setAnswered(false);
  1058→    }
  1059→  }, [currentIndex, questions.length]);
  1060→
  1061→  const handleFinish = useCallback(async () => {
  1062→    if (autoAdvanceRef.current) clearTimeout(autoAdvanceRef.current);
  1063→    if (timerRef.current) clearInterval(timerRef.current);
  1064→
  1065→    const finalElapsed = Math.floor((Date.now() - startTimeRef.current) / 1000);
  1066→    setElapsedSeconds(finalElapsed);
  1067→    setExamState('finished');
  1068→
  1069→    await submitResults(questions, answers, examConfig, finalElapsed);
  1070→  }, [questions, answers, examConfig, submitResults]);
  1071→
  1072→  const handleRestart = useCallback(() => {
  1073→    setExamState('idle');
  1074→    setQuestions([]);
  1075→    setAnswers(new Map());
  1076→    setCurrentIndex(0);
  1077→    setSelectedOption(null);
  1078→    setAnswered(false);
  1079→    setResult(null);
  1080→    setElapsedSeconds(0);
  1081→    setTimeRemaining(0);
  1082→  }, []);
  1083→
  1084→  const handleBack = useCallback(() => {
  1085→    handleRestart();
  1086→  }, [handleRestart]);
  1087→
  1088→  // ─── Render ──────────────────────────────────────────────────────────
  1089→
  1090→  return (
  1091→    <div className="min-h-full">
  1092→      {examState === 'idle' && <IdleState onStart={startExam} />}
  1093→      {examState === 'loading' && <LoadingState />}
  1094→      {examState === 'running' && (
  1095→        <RunningState
  1096→          questions={questions}
  1097→          examConfig={examConfig}
  1098→          timeRemaining={timeRemaining}
  1099→          selectedOption={selectedOption}
  1100→          answered={answered}
  1101→          onSelectOption={handleSelectOption}
  1102→          onNext={handleNext}
  1103→          onFinish={handleFinish}
  1104→          currentIndex={currentIndex}
  1105→          answers={answers}
  1106→        />
  1107→      )}
  1108→      {examState === 'finished' && (
  1109→        <FinishedState
  1110→          questions={questions}
  1111→          answers={answers}
  1112→          result={result}
  1113→          examConfig={examConfig}
  1114→          elapsedSeconds={elapsedSeconds}
  1115→          onRestart={handleRestart}
  1116→          onBack={handleBack}
  1117→        />
  1118→      )}
  1119→    </div>
  1120→  );
  1121→}
  1122→
