   524→export async function POST(request: NextRequest) {
   525→  try {
   526→    const body = await request.json().catch(() => ({}));
   527→    const action = (body as { action?: string }).action;
   528→
   529→    if (action === 'practical') {
   530→      const existing = await db.practicalExercise.count();
   531→      if (existing > 0) {
   532→        return NextResponse.json({
   533→          seeded: true,
   534→          message: `Exercices pratiques déjà présents (${existing})`,
   535→          count: existing,
   536→        });
   537→      }
   538→
   539→      for (const exercise of seedPracticalExercises) {
   540→        await db.practicalExercise.create({ data: exercise });
   541→      }
   542→
   543→      const count = await db.practicalExercise.count();
   544→      return NextResponse.json({
   545→        seeded: true,
   546→        action: 'practical',
   547→        count,
   548→      });
   549→    }
   550→
   551→    if (action === 'v42') {
   552→      const user = await db.user.findFirst({ orderBy: { createdAt: 'asc' } });
   553→      if (!user) {
   554→        return NextResponse.json({ erreur: 'Utilisateur introuvable' }, { status: 404 });
   555→      }
   556→      const v42Counts = await seedV42Data(user.id);
   557→      return NextResponse.json({
   558→        seeded: true,
   559→        action: 'v42',
   560→        counts: v42Counts,
   561→      });
   562→    }
   563→
   564→    // Default: seed everything
   565→    const [existingCountries, existingLicenses, existingSigns, existingQuestions] = await Promise.all([
   566→      db.country.count(),
   567→      db.licenseCategory.count(),
   568→      db.roadSign.count(),
   569→      db.question.count(),
   570→    ]);
   571→
   572→    // Seed countries
   573→    if (existingCountries === 0) {
   574→      for (const country of seedCountries) {
   575→        await db.country.create({ data: country });
   576→      }
   577→    }
   578→
   579→    // Seed licenses
   580→    if (existingLicenses === 0) {
   581→      for (const license of seedLicenseCategories) {
   582→        await db.licenseCategory.create({ data: license });
   583→      }
   584→    }
   585→
   586→    // Seed signs
   587→    if (existingSigns === 0) {
   588→      for (const sign of seedRoadSigns) {
   589→        await db.roadSign.create({ data: sign as Parameters<typeof db.roadSign.create>[0]['data'] });
   590→      }
   591→    }
   592→
   593→    // Seed questions
   594→    if (existingQuestions === 0) {
   595→      for (const question of seedQuestions) {
   596→        await db.question.create({ data: question });
   597→      }
   598→    }
   599→
   600→    // Seed practical exercises
   601→    const existingPractical = await db.practicalExercise.count();
   602→    if (existingPractical === 0) {
   603→      for (const exercise of seedPracticalExercises) {
   604→        await db.practicalExercise.create({ data: exercise });
   605→      }
   606→    }
   607→
   608→    // Seed V4.2 demo data
   609→    const user = await db.user.findFirst({ orderBy: { createdAt: 'asc' } });
   610→    let v42Counts: Record<string, number> = {};
   611→    if (user) {
   612→      v42Counts = await seedV42Data(user.id);
   613→    }
   614→
   615→    const [countries, licenses, signs, questions, practical] = await Promise.all([
   616→      db.country.count(),
   617→      db.licenseCategory.count(),
   618→      db.roadSign.count(),
   619→      db.question.count(),
   620→      db.practicalExercise.count(),
   621→    ]);
   622→
   623→    return NextResponse.json({
   624→      seeded: true,
   625→      counts: { countries, licenses, signs, questions, practical, ...v42Counts },
   626→    });
   627→  } catch (error) {
   628→    console.error('[POST /api/seed] Error:', error);
   629→    return NextResponse.json({ error: 'Seed error' }, { status: 500 });
   630→  }
   631→}
