   245→  const [subCategory, setSubCategory] = useState('');

   259→
   260→    try {
   261→      const res = await fetch('/api/mechanic', {
   262→        method: 'POST',
   263→        headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({ category, subCategory, severity, description }),
   265→      });
   266→
   267→      if (!res.ok) throw new Error('Erreur de diagnostic');
   268→
   269→      const data = await res.json();
   270→      setDiagnosis(data);
   271→    } catch (err) {
   272→      setError(err instanceof Error ? err.message : 'Une erreur est survenue');
   273→    } finally {
   274→      setIsDiagnosing(false);
   275→    }
   276→  };

   278→  const resetForm = () => {

   329→            </div>
   330→          </div>
   331→        </motion.div>

   333→        <Tabs value={activeTab} onValueChange={setActiveTab}>

   336→              <Search className='w-4 h-4 mr-1.5' />
   337→              <span className='hidden sm:inline'>Diagnostic</span>
   338→            </TabsTrigger>
