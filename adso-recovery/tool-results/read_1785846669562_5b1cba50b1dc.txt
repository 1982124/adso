   409→// ─── Animated Number ─────────────────────────────────────────
   410→function AnimatedNumber({ value, className = '' }: { value: number; className?: string }) {
   411→  const [display, setDisplay] = useState(0)
   412→  useEffect(() => {
   413→    let start = 0
   414→    const end = value
   415→    const duration = 1200
   416→    const step = (end - start) / (duration / 16)
   417→    const timer = setInterval(() => {
   418→      start += step
   419→      if ((step > 0 && start >= end) || (step < 0 && start <= end)) {
   420→        setDisplay(end)
   421→        clearInterval(timer)
   422→      } else {
   423→        setDisplay(Math.round(start))
   424→      }
   425→    }, 16)
   426→    return () => clearInterval(timer)
   427→  }, [value])
   428→  return <span className={className}>{display}</span>
   429→}
