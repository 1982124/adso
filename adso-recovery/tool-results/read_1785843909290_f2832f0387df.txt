   169→// ═══════════════════════════════════════════════════════════
   170→// Score Gauge Component
   171→// ═══════════════════════════════════════════════════════════
   172→function ScoreGauge({ score }: { score: number }) {
   173→  const radius = 70;
   174→  const circumference = 2 * Math.PI * radius;
   175→  const progress = (score / 100) * circumference;
   176→  const color = score >= 80 ? '#10b981' : score >= 60 ? '#f59e0b' : '#ef4444';
   177→
   178→  return (
   179→    <div className="relative w-44 h-44 mx-auto">
   180→      <svg className="w-full h-full -rotate-90" viewBox="0 0 160 160">
   181→        <circle
   182→          cx="80" cy="80" r={radius}
   183→          fill="none" stroke="rgba(30,41,59,0.8)" strokeWidth="12"
   184→        />
   185→        <circle
   186→          cx="80" cy="80" r={radius}
   187→          fill="none" stroke={color} strokeWidth="12"
   188→          strokeDasharray={circumference}
   189→          strokeDashoffset={circumference - progress}
   190→          strokeLinecap="round"
   191→          className="transition-all duration-700 ease-out"
   192→        />
   193→      </svg>
   194→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   195→        <span className="text-3xl font-bold text-white">{Math.round(score)}</span>
   196→        <span className="text-xs text-slate-400">/ 100</span>
   197→      </div>
   198→    </div>
   199→  );
   200→}

   202→// ═══════════════════════════════════════════════════════════
   203→// Event Feed Item
   204→// ═══════════════════════════════════════════════════════════
   205→function EventFeedItem({ event }: { event: DrivingEventItem }) {
   206→  const config = severityConfig[event.severity] || severityConfig.normal;
   207→  const Icon = config.icon;
   208→  const label = eventTypeLabels[event.type] || event.type;
   209→
   210→  return (
   211→    <motion.div
   212→      initial={{ opacity: 0, x: -20 }}
   213→      animate={{ opacity: 1, x: 0 }}
   214→      className={`flex items-center gap-3 px-3 py-2 rounded-lg border ${config.bg}`}
   215→    >
   216→      <Icon className={`w-4 h-4 ${config.color}`} />
   217→      <div className="flex-1 min-w-0">
   218→        <div className="flex items-center gap-2">
   219→          <span className="text-sm text-white font-medium">{label}</span>
   220→          <Badge variant="outline" className={`text-[10px] px-1.5 py-0 border-current/30 ${config.color}`}>
   221→            {config.label}
   222→          </Badge>
   223→        </div>
   224→        {event.speed && (
   225→          <span className="text-xs text-slate-400">{event.speed} km/h</span>
   226→        )}
   227→      </div>
   228→      <span className="text-xs text-slate-500 shrink-0">
   229→        {new Date(event.timestamp).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })}
   230→      </span>
   231→    </motion.div>
   232→  );
   233→}

   306→          speed,
   307→          timestamp: new Date().toISOString(),

   340→      await fetch(`/api/driving/sessions/${store.currentSession.id}`, {
   341→        method: 'PATCH',
   342→        headers: { 'Content-Type': 'application/json' },
   343→        body: JSON.stringify({ status: 'paused', duration: store.sessionDuration }),
   344→      });

   385→        status: 'completed',
