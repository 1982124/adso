   480→function shuffleArray<T>(arr: T[]): T[] {
   481→  const shuffled = [...arr];
   482→  for (let i = shuffled.length - 1; i > 0; i--) {
   483→    const j = Math.floor(Math.random() * (i + 1));
   484→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
   485→  }
   486→  return shuffled;
   487→}

   557→        <div className={`${baseClasses} h-16 w-16 ${signColor} rounded-full`}>{children}</div>

   559→    case 'square':
   560→      return (
